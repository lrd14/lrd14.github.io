gurp Lua API Reference (plain text, for pasting into an AI assistant)
api_version: 3
Docs: https://docs.gurp.cc

You are writing Lua 5.4 scripts for gurp, a Roblox external tool. Scripts run in a sandbox.
Use ONLY the functions listed below. Instances are raw integer memory addresses, not Roblox objects.

========================================================================
RULES
========================================================================
- Libraries available: base, coroutine, table, string, math, utf8, os (only clock/time/date/difftime).
- Removed: io, debug, package, require, dofile, loadfile, os.exit/execute/remove/rename/tmpname/getenv/setlocale.
- loadstring(src) / load(src): compile source; returns function or nil, err. Bytecode is refused.
- Long-running scripts MUST call gurp.wait(...) or gurp.sleep_ms(...) (or use RunService callbacks).
  About 50 million instructions are allowed between waits ("script instruction limit" otherwise).
- Limits: 8 scripts at once, 32 MB memory per script, 1 MB source, strings into gurp functions cut at 8192 chars,
  8192 draw commands, 24 gurp.ui windows x 200 elements, http_get bodies up to 2 MB.
- Resetting the Lua environment makes gurp.wait/gurp.sleep_ms raise "Lua environment reset"; the script stops.
- Errors inside gurp functions never crash: bad addresses give nil (getters) / false (setters).
- Lookups (sdk.find_child, sdk.find_child_of_class, sdk.character, get_localplayer...) return 0 when nothing
  is found and nil only when the object passed in was invalid. Check with: if x and x ~= 0 then ... end
- Addresses go stale after respawns/server hops: look objects up again each loop.
- Never hard-code memory offsets; use gurp.get_offset(namespace, name).
- print(...) and warn(...) and gurp.log(msg) write to the gurp Lua console.

Shapes:
- vec3    = { x, y, z }
- matrix3 = { {r00, r01, r02}, {r10, r11, r12}, {r20, r21, r22} }
- cframe  = { r00..r22 (rotation), x, y, z (position) }   -- flat table
- colours in gurp.draw / gurp.ui are separate r, g, b, a numbers from 0 to 1

EXAMPLES

-- Name tags with health over every other player
local hp_off = gurp.get_offset("Humanoid", "Health")
local max_off = gurp.get_offset("Humanoid", "MaxHealth")
local function valid(x) return x and x ~= 0 end
while true do
  gurp.draw.clear()
  local me = gurp.get_localplayer()
  for _, plr in ipairs(gurp.sdk.children(gurp.sdk.players_service() or 0)) do
    local character = plr ~= me and gurp.sdk.character(plr)
    local head = valid(character) and gurp.sdk.find_child(character, "Head")
    local hum = valid(character) and gurp.sdk.find_child_of_class(character, "Humanoid")
    if valid(head) and valid(hum) then
      local p = gurp.sdk.position(head)
      local s = gurp.draw.world_to_screen(p.x, p.y + 1.5, p.z)
      if s then
        local hp = gurp.memory.read_f32(hum + hp_off) or 0
        local max = gurp.memory.read_f32(hum + max_off) or 100
        gurp.draw.text(s.x, s.y, gurp.sdk.display_name(plr), 1, 1, 1, 1, true, 14)
        gurp.draw.rect(s.x - 20, s.y + 16, 40 * math.clamp(hp / math.max(max, 1), 0, 1), 3, 0.3, 1, 0.3, 1)
      end
    end
  end
  gurp.wait(0)
end

-- Small settings window
gurp.ui.window("main", "My Script", 120, 120, 320, 200)
while true do
  local on = gurp.ui.checkbox("main", "on", "Enabled", true)
  local fov = gurp.ui.slider_float("main", "fov", "FOV", 70, 120, 90, 1)
  if on then gurp.sdk.set_field_of_view(gurp.get_camera(), fov) end
  gurp.wait(0)
end
